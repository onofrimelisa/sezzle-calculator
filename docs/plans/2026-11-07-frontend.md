```typescript
// frontend/src/api/client.ts
import axios from 'axios'
import type { ApiErrorResponse } from './types'
import { CalculatorApiError } from './types'

const client = axios.create({
  baseURL: '/api',
  headers: { 'Content-Type': 'application/json' },
  timeout: 10000,
})

client.interceptors.response.use(
  (response) => response,
  (error) => {
    if (axios.isAxiosError(error) && error.response?.data) {
      const data = error.response.data as ApiErrorResponse
      if (data.error) {
        throw new CalculatorApiError(
          data.error.code,
          data.error.message,
          data.error.details
        )
      }
    }
    throw new CalculatorApiError(
      'NETWORK_ERROR',
      'Unable to connect to the server. Please try again.'
    )
  }
)

export default client
```

```typescript
// frontend/src/utils/operations.ts

export interface OperationInfo {
  name: string
  label: string
  symbol: string
  category: 'basic' | 'advanced'
  type: 'binary' | 'unary'
}

export const OPERATIONS: OperationInfo[] = [
  { name: 'add', label: 'Add', symbol: '+', category: 'basic', type: 'binary' },
  { name: 'subtract', label: 'Subtract', symbol: '-', category: 'basic', type: 'binary' },
  { name: 'multiply', label: 'Multiply', symbol: '×', category: 'basic', type: 'binary' },
  { name: 'divide', label: 'Divide', symbol: '÷', category: 'basic', type: 'binary' },
  { name: 'power', label: 'Power', symbol: '^', category: 'advanced', type: 'binary' },
  { name: 'sqrt', label: 'Square Root', symbol: '√', category: 'advanced', type: 'unary' },
  { name: 'percentage', label: 'Percentage', symbol: '%', category: 'advanced', type: 'binary' },
]

export const BASIC_OPERATIONS = OPERATIONS.filter((op) => op.category === 'basic')
export const ADVANCED_OPERATIONS = OPERATIONS.filter((op) => op.category === 'advanced')

export function getOperation(name: string): OperationInfo | undefined {
  return OPERATIONS.find((op) => op.name === name)
}

export function isUnary(operationName: string): boolean {
  return getOperation(operationName)?.type === 'unary'
}

export function formatCalculation(
  operation: string,
  operandA: number,
  operandB: number | undefined,
  result: number
): string {
  const op = getOperation(operation)
  if (!op) return `${operandA} = ${result}`

  if (op.type === 'unary') {
    return `${op.symbol}${operandA} = ${result}`
  }
  return `${operandA} ${op.symbol} ${operandB} = ${result}`
}
```

```typescript
// frontend/src/hooks/useHistory.ts
import { useState, useCallback } from 'react'
import type { CalculateResponse } from '../api/types'

export function useHistory() {
  const [entries, setEntries] = useState<CalculateResponse[]>([])

  const addEntry = useCallback((entry: CalculateResponse) => {
    setEntries((prev) => [entry, ...prev])
  }, [])

  const clearHistory = useCallback(() => {
    setEntries([])
  }, [])

  return { entries, addEntry, clearHistory }
}
```

```typescript
// frontend/src/hooks/useCalculator.ts
import { useState, useCallback } from 'react'
import { calculate } from '../api/calculator'
import type { CalculateResponse } from '../api/types'
import { CalculatorApiError } from '../api/types'

export function useCalculator() {
  const [selectedOperation, setSelectedOperation] = useState('add')
  const [result, setResult] = useState<CalculateResponse | null>(null)
  const [error, setError] = useState<string | null>(null)
  const [isLoading, setIsLoading] = useState(false)

  const selectOperation = useCallback((operation: string) => {
    setSelectedOperation(operation)
  }, [])

  const performCalculation = useCallback(
    async (operandA: number, operandB?: number) => {
      setIsLoading(true)
      setError(null)
      setResult(null)

      try {
        const response = await calculate({
          operation: selectedOperation,
          operand_a: operandA,
          operand_b: operandB,
        })
        setResult(response)
        return response
      } catch (err) {
        if (err instanceof CalculatorApiError) {
          setError(err.message)
        } else {
          setError('An unexpected error occurred')
        }
        return null
      } finally {
        setIsLoading(false)
      }
    },
    [selectedOperation]
  )

  return {
    selectedOperation,
    selectOperation,
    result,
    error,
    isLoading,
    performCalculation,
  }
}
```

```tsx
// frontend/src/components/Calculator.tsx
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { useCalculator } from '../hooks/useCalculator'
import { useHistory } from '../hooks/useHistory'
import { isUnary } from '../utils/operations'
import { calculatorSchema } from '../schemas/calculator'
import { operandSchema } from '../schemas/calculator'
import type { CalculatorFormData } from '../schemas/calculator'
import { Display } from './Display'
import { OperandInputs } from './OperandInputs'
import { OperationButtons } from './OperationButtons'
import { History } from './History'
import { useEffect } from 'react'

export function Calculator() {
  const { selectedOperation, selectOperation, result, error, isLoading, performCalculation } =
    useCalculator()
  const { entries, addEntry, clearHistory } = useHistory()

  const {
    register,
    handleSubmit,
    setValue,
    formState: { errors },
  } = useForm<CalculatorFormData>({
    defaultValues: { operandA: '', operandB: '' },
  })

  const unary = isUnary(selectedOperation)

  useEffect(() => {
    if (unary) {
      setValue('operandB', '')
    }
  }, [unary, setValue])

  const onSelectOperation = (operation: string) => {
    selectOperation(operation)
  }

  const onSubmit = async () => {
    const operandAInput = document.getElementById('operandA') as HTMLInputElement
    const operandBInput = document.getElementById('operandB') as HTMLInputElement

    const parsedA = operandSchema.safeParse(operandAInput?.value)
    if (!parsedA.success) return

    let operandB: number | undefined
    if (!unary) {
      const parsedB = operandSchema.safeParse(operandBInput?.value)
      if (!parsedB.success) return
      operandB = parsedB.data
    }

    const response = await performCalculation(parsedA.data, operandB)
    if (response) {
      addEntry(response)
    }
  }

  return (
    <div className="mx-auto w-full max-w-md space-y-4 p-4">
      <Display result={result?.result ?? null} error={error} isLoading={isLoading} />
      <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
        <OperandInputs register={register} errors={errors} isUnary={unary} />
        <OperationButtons selectedOperation={selectedOperation} onSelect={onSelectOperation} />
        <button
          type="submit"
          disabled={isLoading}
          className="w-full rounded-lg bg-green-600 p-3 text-lg font-bold text-white transition-colors hover:bg-green-700 disabled:opacity-50"
        >
          {isLoading ? 'Calculating...' : 'Calculate'}
        </button>
      </form>
      <History entries={entries} onClear={clearHistory} />
    </div>
  )
}
```

```tsx
// frontend/src/components/ErrorBoundary.tsx
import { Component } from 'react'
import type { ReactNode, ErrorInfo } from 'react'

interface Props {
  children: ReactNode
}

interface State {
  hasError: boolean
  error: Error | null
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false, error: null }
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="flex min-h-screen items-center justify-center bg-gray-50 p-4">
          <div className="rounded-lg bg-white p-8 text-center shadow-lg">
            <h1 className="mb-2 text-2xl font-bold text-red-600">Something went wrong</h1>
            <p className="mb-4 text-gray-600">{this.state.error?.message}</p>
            <button
              onClick={() => this.setState({ hasError: false, error: null })}
              className="rounded-lg bg-blue-600 px-4 py-2 text-white hover:bg-blue-700"
            >
              Try again
            </button>
          </div>
        </div>
      )
    }

    return this.props.children
  }
}
```
