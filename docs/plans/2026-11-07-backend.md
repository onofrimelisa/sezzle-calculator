// backend/internal/domain/registry.go
package domain

// RegistryEntry holds a registered operation with its type and metadata.
type RegistryEntry struct {
	Type        OperationType
	Name        string
	Description string
	Category    string
	Binary      BinaryOperation
	Unary       UnaryOperation
}

// Registry stores all available operations indexed by name.
type Registry struct {
	operations map[string]RegistryEntry
}

func NewRegistry() *Registry {
	return &Registry{
		operations: make(map[string]RegistryEntry),
	}
}

func (r *Registry) RegisterBinary(op BinaryOperation) {
	r.operations[op.Name()] = RegistryEntry{
		Type:        Binary,
		Name:        op.Name(),
		Description: op.Description(),
		Category:    op.Category(),
		Binary:      op,
	}
}

func (r *Registry) RegisterUnary(op UnaryOperation) {
	r.operations[op.Name()] = RegistryEntry{
		Type:        Unary,
		Name:        op.Name(),
		Description: op.Description(),
		Category:    op.Category(),
		Unary:       op,
	}
}

func (r *Registry) Get(name string) (RegistryEntry, error) {
	entry, ok := r.operations[name]
	if !ok {
		return RegistryEntry{}, ErrInvalidOperation
	}
	return entry, nil
}

func (r *Registry) List() []RegistryEntry {
	entries := make([]RegistryEntry, 0, len(r.operations))
	for _, entry := range r.operations {
		entries = append(entries, entry)
	}
	return entries
}
```

// backend/internal/platform/config/config.go
package config

import (
	"os"
	"strconv"
	"time"
)

type Config struct {
	Port         int
	LogLevel     string
	CORSOrigins  string
	ReadTimeout  time.Duration
	WriteTimeout time.Duration
}

func Load() Config {
	return Config{
		Port:         getEnvInt("PORT", 8080),
		LogLevel:     getEnvStr("LOG_LEVEL", "info"),
		CORSOrigins:  getEnvStr("CORS_ORIGINS", "*"),
		ReadTimeout:  getEnvDuration("READ_TIMEOUT", 5*time.Second),
		WriteTimeout: getEnvDuration("WRITE_TIMEOUT", 10*time.Second),
	}
}

func getEnvStr(key, fallback string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return fallback
}

func getEnvInt(key string, fallback int) int {
	if v := os.Getenv(key); v != "" {
		if i, err := strconv.Atoi(v); err == nil {
			return i
		}
	}
	return fallback
}

func getEnvDuration(key string, fallback time.Duration) time.Duration {
	if v := os.Getenv(key); v != "" {
		if d, err := time.ParseDuration(v); err == nil {
			return d
		}
	}
	return fallback
}
```

// backend/internal/platform/logger/logger.go
package logger

import (
	"io"
	"log/slog"
	"os"
	"strings"
)

func New(level string, w ...io.Writer) *slog.Logger {
	var writer io.Writer = os.Stdout
	if len(w) > 0 {
		writer = w[0]
	}

	lvl := parseLevel(level)
	opts := &slog.HandlerOptions{Level: lvl}
	handler := slog.NewJSONHandler(writer, opts)

	return slog.New(handler)
}

func parseLevel(level string) slog.Level {
	switch strings.ToLower(level) {
	case "debug":
		return slog.LevelDebug
	case "warn":
		return slog.LevelWarn
	case "error":
		return slog.LevelError
	default:
		return slog.LevelInfo
	}
}
```

// backend/internal/dto/calculator.go
package dto

import "time"

// CalculateRequest represents the incoming calculation request.
type CalculateRequest struct {
	Operation string   `json:"operation" validate:"required"`
	OperandA  float64  `json:"operand_a" validate:"required"`
	OperandB  *float64 `json:"operand_b,omitempty"`
}

// CalculateResponse represents a successful calculation result.
type CalculateResponse struct {
	Operation string   `json:"operation"`
	OperandA  float64  `json:"operand_a"`
	OperandB  *float64 `json:"operand_b,omitempty"`
	Result    float64  `json:"result"`
	Timestamp string   `json:"timestamp"`
}

func NewCalculateResponse(operation string, a float64, b *float64, result float64) CalculateResponse {
	return CalculateResponse{
		Operation: operation,
		OperandA:  a,
		OperandB:  b,
		Result:    result,
		Timestamp: time.Now().UTC().Format(time.RFC3339),
	}
}

// ErrorDetail holds the structured error information.
type ErrorDetail struct {
	Code    string                 `json:"code"`
	Message string                 `json:"message"`
	Details map[string]interface{} `json:"details,omitempty"`
}

// ErrorResponse wraps an error detail for JSON responses.
type ErrorResponse struct {
	Error ErrorDetail `json:"error"`
}

func NewErrorResponse(code, message string, details map[string]interface{}) ErrorResponse {
	return ErrorResponse{
		Error: ErrorDetail{
			Code:    code,
			Message: message,
			Details: details,
		},
	}
}
```

// backend/internal/service/calculator.go
package service

import (
	"fmt"
	"log/slog"

	"github.com/sezzle/calculator-backend/internal/domain"
	"github.com/sezzle/calculator-backend/internal/dto"
	"github.com/sezzle/calculator-backend/internal/platform/metrics"
)

type CalculatorService struct {
	registry *domain.Registry
	logger   *slog.Logger
	metrics  *metrics.Metrics
}

func NewCalculatorService(registry *domain.Registry, logger *slog.Logger, m *metrics.Metrics) *CalculatorService {
	return &CalculatorService{
		registry: registry,
		logger:   logger,
		metrics:  m,
	}
}

func (s *CalculatorService) Calculate(req dto.CalculateRequest) (dto.CalculateResponse, error) {
	entry, err := s.registry.Get(req.Operation)
	if err != nil {
		s.logger.Warn("invalid operation requested", "operation", req.Operation)
		return dto.CalculateResponse{}, err
	}

	var result float64

	switch entry.Type {
	case domain.Binary:
		if req.OperandB == nil {
			return dto.CalculateResponse{}, domain.ErrMissingOperand
		}
		result, err = entry.Binary.Execute(req.OperandA, *req.OperandB)

	case domain.Unary:
		if req.OperandB != nil {
			return dto.CalculateResponse{}, domain.ErrUnexpectedOperand
		}
		result, err = entry.Unary.Execute(req.OperandA)

	default:
		return dto.CalculateResponse{}, fmt.Errorf("unknown operation type: %s", entry.Type)
	}

	if err != nil {
		s.logger.Warn("operation failed",
			"operation", req.Operation,
			"operand_a", req.OperandA,
			"error", err.Error(),
		)
		return dto.CalculateResponse{}, err
	}

	s.logger.Debug("calculation successful",
		"operation", req.Operation,
		"operand_a", req.OperandA,
		"result", result,
	)

	return dto.NewCalculateResponse(req.Operation, req.OperandA, req.OperandB, result), nil
}

const requestIDKey contextKey = "request_id"

func RequestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		id := uuid.New().String()

	if id, ok := ctx.Value(requestIDKey).(string); ok {

	}

// backend/internal/handler/calculator.go
package handler

import (
	"encoding/json"
	"errors"
	"log/slog"
	"net/http"

	"github.com/go-playground/validator/v10"
	"github.com/sezzle/calculator-backend/internal/domain"
	"github.com/sezzle/calculator-backend/internal/dto"
	"github.com/sezzle/calculator-backend/internal/service"
)

type CalculatorHandler struct {
	service   *service.CalculatorService
	validator *validator.Validate
	logger    *slog.Logger
}

func NewCalculatorHandler(svc *service.CalculatorService, v *validator.Validate, logger *slog.Logger) *CalculatorHandler {
	return &CalculatorHandler{
		service:   svc,
		validator: v,
		logger:    logger,
	}
}

// Calculate godoc
// @Summary Perform a calculation
// @Description Execute an arithmetic operation with one or two operands
// @Tags Calculator
// @Accept json
// @Produce json
// @Param request body dto.CalculateRequest true "Calculation request"
// @Success 200 {object} dto.CalculateResponse
// @Failure 400 {object} dto.ErrorResponse
// @Failure 422 {object} dto.ErrorResponse
// @Router /api/calculate [post]
func (h *CalculatorHandler) Calculate(w http.ResponseWriter, r *http.Request) {
	var req dto.CalculateRequest

	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		h.writeError(w, http.StatusUnprocessableEntity, "INVALID_INPUT", "Invalid JSON payload", nil)
		return
	}

	if err := h.validator.Struct(req); err != nil {
		h.writeError(w, http.StatusUnprocessableEntity, "INVALID_INPUT", "Validation failed: "+err.Error(), nil)
		return
	}

	resp, err := h.service.Calculate(req)
	if err != nil {
		h.handleServiceError(w, err, req)
		return
	}

	h.writeJSON(w, http.StatusOK, resp)
}

func (h *CalculatorHandler) handleServiceError(w http.ResponseWriter, err error, req dto.CalculateRequest) {
	details := map[string]interface{}{
		"operation": req.Operation,
		"operand_a": req.OperandA,
	}
	if req.OperandB != nil {
		details["operand_b"] = *req.OperandB
	}

	var domErr *domain.DomainError
	if errors.As(err, &domErr) {
		status := http.StatusUnprocessableEntity
		if domErr.Code == "INVALID_OPERATION" {
			status = http.StatusBadRequest
		}
		h.writeError(w, status, domErr.Code, domErr.Message, details)
		return
	}

	h.logger.Error("unexpected error", "error", err.Error())
	h.writeError(w, http.StatusInternalServerError, "INTERNAL_ERROR", "An unexpected error occurred", nil)
}

func (h *CalculatorHandler) writeJSON(w http.ResponseWriter, status int, data interface{}) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(data)
}

func (h *CalculatorHandler) writeError(w http.ResponseWriter, status int, code, message string, details map[string]interface{}) {
	h.writeJSON(w, status, dto.NewErrorResponse(code, message, details))
}

package main

import (
	"context"

	"syscall"

	"github.com/prometheus/client_golang/prometheus/promhttp"
	httpSwagger "github.com/swaggo/http-swagger/v2"

	"github.com/sezzle/calculator-backend/internal/domain"
	"github.com/sezzle/calculator-backend/internal/handler"
	"github.com/sezzle/calculator-backend/internal/middleware"
	"github.com/sezzle/calculator-backend/internal/operations"
	"github.com/sezzle/calculator-backend/internal/platform/config"
	"github.com/sezzle/calculator-backend/internal/platform/logger"
	"github.com/sezzle/calculator-backend/internal/platform/metrics"
	"github.com/sezzle/calculator-backend/internal/service"

	_ "github.com/sezzle/calculator-backend/docs"
)

// @title Calculator API
// @version 1.0
// @description A RESTful calculator API with basic and advanced arithmetic operations.
// @host localhost:8080
// @BasePath /

func main() {
	cfg := config.Load()
	log := logger.New(cfg.LogLevel)
	m := metrics.New()

	// Build registry
	registry := buildRegistry()

	// Build service
	svc := service.NewCalculatorService(registry, log, m)

	// Build handler
	v := validator.New()
	calcHandler := handler.NewCalculatorHandler(svc, v, log)

	// Build router
	r := chi.NewRouter()

	// Middleware stack
	r.Use(middleware.Recovery(log))
	r.Use(middleware.RequestID)
	r.Use(middleware.Logging(log))
	r.Use(middleware.Metrics(m))
	r.Use(cors.Handler(cors.Options{
		AllowedOrigins:   []string{cfg.CORSOrigins},
		AllowedMethods:   []string{"GET", "POST", "OPTIONS"},
		AllowedHeaders:   []string{"Accept", "Content-Type"},
		ExposedHeaders:   []string{"X-Request-ID"},
		AllowCredentials: false,
		MaxAge:           300,
	}))

	// Routes
	r.Post("/api/calculate", calcHandler.Calculate)
	r.Get("/healthz", handler.HealthCheck)
	r.Get("/metrics", promhttp.Handler().ServeHTTP)
	r.Get("/swagger/*", httpSwagger.WrapHandler)

	// Server
	srv := &http.Server{
		Addr:         fmt.Sprintf(":%d", cfg.Port),
		Handler:      r,
		ReadTimeout:  cfg.ReadTimeout,
		WriteTimeout: cfg.WriteTimeout,
	}

	// Graceful shutdown
	go func() {
		log.Info("server starting", "port", cfg.Port)
		if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Error("server failed", "error", err.Error())
			os.Exit(1)
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	<-quit

	log.Info("server shutting down...")

	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	if err := srv.Shutdown(ctx); err != nil {
		log.Error("server forced shutdown", "error", err.Error())
		os.Exit(1)
	}

	log.Info("server stopped")
}

	registry.RegisterBinary(operations.NewMultiply())

	// Advanced operations
	registry.RegisterBinary(operations.NewPower())
	registry.RegisterUnary(operations.NewSqrt())
	registry.RegisterBinary(operations.NewPercentage())

#!/usr/bin/env bash
# backend/scripts/analyze.sh
# Validates a spec for completeness

set -euo pipefail

OP_NAME="${1:?Usage: analyze.sh <operation_name>}"
SPEC_FILE="spec/operations/${OP_NAME}.md"

if [ ! -f "$SPEC_FILE" ]; then
    echo "Error: Spec not found at ${SPEC_FILE}"
    exit 1
fi

ERRORS=0

check_section() {
    local section="$1"
    if ! grep -q "## ${section}" "$SPEC_FILE"; then
        echo "MISSING: ## ${section} section"
        ERRORS=$((ERRORS + 1))
    fi
}

check_not_placeholder() {
    local pattern="$1"
    local label="$2"
    if grep -q "$pattern" "$SPEC_FILE"; then
        echo "INCOMPLETE: ${label} still contains placeholder text"
        ERRORS=$((ERRORS + 1))
    fi
}

echo "Analyzing spec: ${SPEC_FILE}"
echo "---"

check_section "Description"
check_section "Category"
check_section "Type"
check_section "Inputs"
check_section "Output"
check_section "Edge Cases"
check_section "Examples"
check_section "Test Cases"
check_section "Swagger"

check_not_placeholder "Brief description" "Description"
check_not_placeholder "one-line summary" "Swagger Summary"
check_not_placeholder "list applicable" "Swagger Error responses"

# Check at least one example has a result
if grep -q "| ? " "$SPEC_FILE"; then
    echo "INCOMPLETE: Examples table has unfilled results (?)"
    ERRORS=$((ERRORS + 1))
fi

# Check at least one test case is defined (not just placeholders)
TESTS=$(grep -c "\- \[ \]" "$SPEC_FILE" || true)
if [ "$TESTS" -lt 3 ]; then
    echo "WARNING: Only ${TESTS} test cases defined (recommend at least 3)"
fi

echo "---"
if [ "$ERRORS" -gt 0 ]; then
    echo "FAILED: ${ERRORS} issue(s) found. Fix them before running: make apply op=${OP_NAME}"
    exit 1
else
    echo "PASSED: Spec is complete. Run: make apply op=${OP_NAME}"

#!/usr/bin/env bash
# backend/scripts/sync.sh
# Regenerates swagger and validates spec-code alignment

set -euo pipefail

echo "Regenerating Swagger docs..."
swag init -g cmd/server/main.go -o docs
echo "Swagger docs regenerated."

echo ""
echo "Checking spec-code alignment..."

SPECS_DIR="spec/operations"
OPS_DIR="internal/operations"
ERRORS=0

for spec in "${SPECS_DIR}"/*.md; do
    [ -f "$spec" ] || continue
    op_name=$(basename "$spec" .md)
    if [ ! -f "${OPS_DIR}/${op_name}.go" ]; then
        echo "MISSING IMPLEMENTATION: ${op_name} (spec exists, code missing)"
        ERRORS=$((ERRORS + 1))
    fi
done

for impl in "${OPS_DIR}"/*.go; do
    [ -f "$impl" ] || continue
    op_name=$(basename "$impl" .go)
    [ "$op_name" = "doc" ] && continue  # skip doc.go if any
    if [ ! -f "${SPECS_DIR}/${op_name}.md" ]; then
        echo "MISSING SPEC: ${op_name} (code exists, spec missing)"
    fi
done

if [ "$ERRORS" -gt 0 ]; then
    echo "SYNC FAILED: ${ERRORS} spec(s) without implementation"
    exit 1
else
    echo "SYNC OK: All specs have implementations"
